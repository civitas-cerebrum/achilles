import type { Steps } from '@civitas-cerebrum/element-interactions';

export async function createAccount(steps: Steps, maxRetries: number) {
  let accountCreated = false;
  while (true) { // EXPECT TB-X4/advisory
    await steps.click('createButton', 'SignupPage');
    accountCreated = await steps.isVisible('welcomeBanner', 'DashboardPage');
    if (accountCreated || maxRetries > 3) break;
  }
  await steps.verifyPresence('welcomeBanner', 'DashboardPage');
}
