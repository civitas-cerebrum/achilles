import type { Steps } from '@civitas-cerebrum/element-interactions';

export async function loadWholeFeed(steps: Steps) {
  while (await steps.isVisible('loadMoreButton', 'FeedPage')) { // EXPECT TB-X4/advisory
    await steps.click('loadMoreButton', 'FeedPage');
  }
  await steps.verifyPresence('feedEnd', 'FeedPage');
}
