/** File-local: only the suspension tests sign in as a suspended account. */
async function signInSuspended(steps: Steps) {
  await steps.click('signInLink', 'HeaderPage');
}

test('TCST-000301 · a suspended account sees the notice', async ({ steps }) => {

  await steps.verifyPresence('suspendedNotice', 'HomePage');
});
