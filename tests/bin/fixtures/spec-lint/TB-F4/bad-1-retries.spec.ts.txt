test('TCAB-0040 · retried in the test', async ({ steps }) => {
  await steps.retryUntil(async () => { await steps.click('refresh', 'InboxPage'); }, async () => { await steps.verifyPresence('newMail', 'InboxPage'); }, 3, 1000); // EXPECT TB-F4

    if (await steps.isVisible('newMail', 'InboxPage')) break;
    await steps.click('refresh', 'InboxPage');

  await steps.verifyPresence('newMail', 'InboxPage');
});
