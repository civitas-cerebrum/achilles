import type { Steps } from '@civitas-cerebrum/element-interactions';

export async function dismissCookieBanner(steps: Steps) {
  await steps.clickIfPresent('acceptCookies', 'HomePage');
  await steps.on('promo', 'HomePage').ifVisible().click();
  await steps.waitForState('banner', 'HomePage', 'hidden', { optional: true });
}
