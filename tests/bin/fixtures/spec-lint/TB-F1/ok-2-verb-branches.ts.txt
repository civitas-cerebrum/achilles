import type { Steps } from '@civitas-cerebrum/element-interactions';

export async function dismissPromo(steps: Steps) {
  if (await steps.isVisible('promoBanner', 'HomePage')) await steps.click('closePromo', 'HomePage');
  await steps.verifyAbsence('promoBanner', 'HomePage');
}
