let lastUser: any;
let n = 0;
let token;
export function setUser(u) { lastUser = u; }
export function bump(k) { n = k; }
export function setToken(t) { token = t; }
export async function deleteLastUser(steps) {
  const id = String(lastUser.id).trim();
  const label = n.toFixed(0);
  const auth = token.trim();
  await steps.apiDelete(`/users/${id}?n=${label}`, { headers: { Authorization: auth } });
}
export class UserApi {
  constructor(private steps: any) {}
  async remove() { const id = lastUser.id.toString(); await this.steps.apiDelete(`/users/${id}`); }
}
