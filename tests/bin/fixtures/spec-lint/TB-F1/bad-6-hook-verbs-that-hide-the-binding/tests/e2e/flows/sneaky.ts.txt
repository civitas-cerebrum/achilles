export async function aliasClose(steps) {
  const s = steps;
  await s.click('closePromo', 'HomePage');
}
export async function destructClose(steps) {
  const { click } = steps;
  await click('closePromo', 'HomePage');
}
export async function viaApiResult(steps) {
  const r = await steps.apiGet('/promo');
  const act = r.actions.close;
  await act();
}
export async function optsBag(opts) {
  await opts.steps.click('closePromo', 'HomePage');
}
export async function apiOnlyReal(steps) {
  const r = await steps.apiDelete('/accounts/u');
  return String(await r.json());
}
