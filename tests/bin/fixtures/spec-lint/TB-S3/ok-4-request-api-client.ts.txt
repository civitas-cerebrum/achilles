import type { APIRequestContext } from '@playwright/test';

export class OrdersApi {
  constructor(private request: APIRequestContext) {}
  async create(body: Record<string, unknown>) {
    return this.request.post('/api/orders', { data: body });
  }
  async get(id: string) {
    return this.request.get(`/api/orders/${id}`);
  }
}

export class CustomersApi {
  private readonly http: APIRequestContext;
  constructor(request: APIRequestContext) {
    this.http = request;
  }
  async remove(id: string) {
    await this.http.delete(`/api/customers/${id}`);
  }
}

export function ordersFor(request: APIRequestContext) {
  return new OrdersApi(request);
}
