import type { Steps } from '@civitas-cerebrum/element-interactions';

export class ListingLedger {
  private ids: string[] = [];
  add(id: string) { this.ids.push(id); }
  async clear(x: { steps: Steps }) { for (const id of this.ids) await x.steps.apiDelete(`/listings/${id}`); }
}
export function ledgerFor(steps: Steps) { return new ListingLedger(); }
